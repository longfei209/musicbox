const input = document.getElementById('kwInput');
const searchBtn = document.getElementById('searchBtn');
const saveBtn = document.getElementById('saveBtn');
const clearBtn = document.getElementById('clearBtn');
const status = document.getElementById('status');

function parseKeywords(raw) {
  return [...new Set(
    raw.split(/[\s,，、]+/)
      .map(s => s.trim())
      .filter(Boolean)
  )];
}

function send(msg) {
  return new Promise(resolve => {
    chrome.tabs.query({ active: true, currentWindow: true }, tabs => {
      const tabId = tabs[0]?.id;
      if (!tabId) return resolve(null);
      chrome.tabs.sendMessage(tabId, msg, resp => {
        if (chrome.runtime.lastError) return resolve(null);
        resolve(resp);
      });
    });
  });
}

// 打开面板时载入默认词，自动聚焦并全选，直接回车即可搜默认词
chrome.storage.local.get({ keywords: [] }, data => {
  input.value = data.keywords.join(' ');
  input.focus();
  input.select();
});

async function doSearch() {
  const kws = parseKeywords(input.value);
  if (!kws.length) { status.textContent = '请输入关键词'; return; }
  const resp = await send({ type: 'SEARCH', keywords: kws });
  status.textContent = resp
    ? `已标记 ${resp.count} 个框`
    : '当前页面不支持标记';
}

async function doSave() {
  const kws = parseKeywords(input.value);
  if (!kws.length) { status.textContent = '请输入关键词'; return; }
  chrome.storage.local.set({ keywords: kws }, async () => {
    const resp = await send({ type: 'SEARCH', keywords: kws });
    status.textContent = `已保存默认词，标记 ${resp ? resp.count : 0} 个框`;
  });
}

async function doClear() {
  await send({ type: 'CLEAR' });
  status.textContent = '已清除标记';
}

searchBtn.onclick = doSearch;
saveBtn.onclick = doSave;
clearBtn.onclick = doClear;

// Enter 搜索，中文输入法选词时不误触发
input.addEventListener('keydown', e => {
  if (e.key === 'Enter' && !e.isComposing) {
    e.preventDefault();
    doSearch();
  }
});

文件：content.js

let tags = [];

function clearMarks() {
  document.querySelectorAll('[data-mh-mark]').forEach(el => {
    el.style.outline = '';
    el.style.outlineOffset = '';
    el.removeAttribute('data-mh-mark');
  });
  document.querySelectorAll('[data-mh-tag]').forEach(el => el.remove());
  tags = [];
  window.removeEventListener('scroll', updateTags, true);
  window.removeEventListener('resize', updateTags, true);
}

function updateTags() {
  tags.forEach(t => {
    const r = t.el.getBoundingClientRect();
    if (r.bottom < 0 || r.top > window.innerHeight) {
      t.tag.style.display = 'none';
      return;
    }
    t.tag.style.display = '';
    t.tag.style.left = Math.max(4, r.left) + 'px';
    t.tag.style.top = Math.max(4, r.top - 20) + 'px';
  });
}

function markInputs(keywords) {
  clearMarks();
  if (!keywords.length) return 0;

  const skip = [
    'hidden', 'password', 'file', 'checkbox', 'radio',
    'submit', 'button', 'reset', 'image', 'range',
    'color', 'date', 'time', 'month', 'week'
  ];

  document.querySelectorAll('input, textarea').forEach(el => {
    const type = (el.type || 'text').toLowerCase();
    if (skip.indexOf(type) !== -1 || el.disabled) return;

    const v = el.value || '';
    if (!v) return;

    const lv = v.toLowerCase();
    const matched = keywords.filter(kw => lv.indexOf(kw.toLowerCase()) !== -1);
    if (!matched.length) return;

    el.setAttribute('data-mh-mark', '1');
    el.style.outline = '2px solid #ff8a00';
    el.style.outlineOffset = '1px';

    const tag = document.createElement('div');
    tag.setAttribute('data-mh-tag', '1');
    tag.textContent = matched.join(' · ');
    tag.style.cssText =
      'position:fixed;background:#ff8a00;color:#fff;font-size:11px;' +
      'line-height:16px;padding:0 6px;border-radius:8px;' +
      'z-index:2147483646;pointer-events:none;font-family:system-ui;' +
      'white-space:nowrap;max-width:220px;overflow:hidden;text-overflow:ellipsis';
    document.body.appendChild(tag);

    tags.push({ el, tag });
  });

  if (tags.length) {
    updateTags();
    window.addEventListener('scroll', updateTags, true);
    window.addEventListener('resize', updateTags, true);
  }

  return tags.length;
}

chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
  if (msg.type === 'SEARCH') {
    const count = markInputs(msg.keywords || []);
    sendResponse({ count });
  } else if (msg.type === 'CLEAR') {
    clearMarks();
    sendResponse({ count: 0 });
  }
  return true;
});

// 页面加载时自动应用默认词
chrome.storage.local.get({ keywords: [] }, data => {
  if (!data.keywords.length) return;

  const apply = () => setTimeout(() => markInputs(data.keywords), 300);

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', apply);
  } else {
    apply();
  }
});


================================
安装方法
================================
1. 新建文件夹：input-marker
2. 把上面 4 个文件分别保存进去：
   - manifest.json
   - popup.html
   - popup.js
   - content.js
3. 打开 Edge，地址栏输入：
   edge://extensions/
4. 打开左下角“开发人员模式”
5. 点击“加载解压缩的扩展”
6. 选择 input-marker 文件夹
7. 按 Ctrl+Shift+F 打开面板


================================
使用说明
================================
【搜索】
- 打开面板后，输入框自动全选默认词
- 直接按 Enter → 立即搜索并标记当前页
- 想换词 → 直接打字覆盖，再按 Enter

【设为默认】
- 输入框里的词会保存下来
- 每次打开新网页，自动用这些词标记输入框
- 保存的同时也会立即标记当前页

【清除】
- 去掉当前页所有橙色边框和标签

【分隔方式】
- 空格、逗号、顿号都行，例如：
  报错 警告 TODO
  报错, 警告, TODO
  报错、警告、TODO

【标记效果】
- 有匹配的输入框：橙色边框 + 右上角标签写明命中了哪些词
- 标签跟着页面滚动自动调整，滚出屏幕自动隐藏
- 面板底部显示“已标记 X 个框”


================================
注意事项
================================
- 快捷键 Ctrl+Shift+F 如果冲突，去 edge://extensions/shortcuts 改
- edge:// 页面、PDF、部分内置页无法注入，不会显示标记
- 扩展只标记“已经有内容”的输入框，空输入框不标
- 动态加载的输入框不会自动标记，刷新页面或重新搜索即可
- 修改代码后，要在 edge://extensions/ 里点扩展的“刷新”，再刷新网页
