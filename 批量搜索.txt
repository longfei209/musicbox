(function(){
  var raw = prompt('关键词，逗号或换行分隔');
  if(!raw) return;
  var kws = raw.split(/[,，\n]/).map(function(s){return s.trim()}).filter(Boolean);
  if(!kws.length) return;

  document.querySelectorAll('[data-mh-input]').forEach(function(el){
    el.style.outline = ''; el.style.outlineOffset = ''; el.removeAttribute('data-mh-input');
  });
  document.querySelectorAll('[data-mh-label]').forEach(function(el){ el.remove(); });
  var ob = document.getElementById('__mh_bar'); if(ob) ob.remove();

  var allHits = [];
  var skip = ['hidden','password','file','checkbox','radio','submit','button','reset','image','range','color','date','time','month','week'];

  document.querySelectorAll('input, textarea').forEach(function(el){
    var type = (el.type || 'text').toLowerCase();
    if(skip.indexOf(type) !== -1 || el.disabled) return;
    var v = el.value || ''; if(!v) return;

    var lv = v.toLowerCase(), boxHits = [];
    kws.forEach(function(kw){
      var lk = kw.toLowerCase(), i = lv.indexOf(lk);
      while(i !== -1){ boxHits.push({s:i, e:i+kw.length, kw:kw}); i = lv.indexOf(lk, i + lk.length); }
    });
    if(!boxHits.length) return;

    boxHits.sort(function(a,b){ return a.s - b.s; });
    var f = [], last = -1;
    boxHits.forEach(function(h){ if(h.s >= last){ f.push(h); last = h.e; } });

    el.setAttribute('data-mh-input','1');
    el.style.outline = '2px solid #ff5252';
    el.style.outlineOffset = '1px';
    f.forEach(function(h){ allHits.push({ el:el, s:h.s, e:h.e }); });

    var r = el.getBoundingClientRect();
    var top = r.top - 20; if(top < 4) top = r.bottom + 4;
    var lb = document.createElement('div');
    lb.setAttribute('data-mh-label','1');
    lb.textContent = f.length + ' 处';
    lb.style.cssText = 'position:fixed;left:' + Math.max(4, r.right - 46) + 'px;top:' + top + 'px;background:#ff5252;color:#fff;font-size:11px;line-height:16px;padding:0 6px;border-radius:8px;z-index:2147483647;pointer-events:none;font-family:system-ui;white-space:nowrap';
    document.body.appendChild(lb);
  });

  if(!allHits.length){ alert('输入框里没有找到匹配'); return; }

  var cur = 0;

  function go(i){
    if(i < 0) i = allHits.length - 1;
    if(i >= allHits.length) i = 0;
    cur = i;
    var h = allHits[i];
    h.el.focus();
    try { h.el.setSelectionRange(h.s, h.e); } catch(e){}
    h.el.scrollIntoView({ block:'center', behavior:'smooth' });
    document.getElementById('__mh_info').textContent = (cur+1) + ' / ' + allHits.length;
  }

  var bar = document.createElement('div');
  bar.id = '__mh_bar';
  bar.style.cssText = 'position:fixed;bottom:20px;left:50%;transform:translateX(-50%);background:#222;color:#fff;padding:6px 12px;border-radius:20px;z-index:2147483647;font-family:system-ui;font-size:13px;box-shadow:0 2px 10px rgba(0,0,0,.3);display:flex;align-items:center;gap:8px';

  var info = document.createElement('span');
  info.id = '__mh_info';
  bar.appendChild(info);

  var pb = document.createElement('button');
  pb.textContent = '上一个';
  pb.style.cssText = 'background:#444;color:#fff;border:none;padding:3px 8px;border-radius:10px;cursor:pointer;font-size:12px;font-family:inherit';
  pb.onclick = function(){ go(cur - 1); };
  bar.appendChild(pb);

  var nb = document.createElement('button');
  nb.textContent = '下一个';
  nb.style.cssText = 'background:#0067c0;color:#fff;border:none;padding:3px 8px;border-radius:10px;cursor:pointer;font-size:12px;font-family:inherit';
  nb.onclick = function(){ go(cur + 1); };
  bar.appendChild(nb);

  var cb = document.createElement('button');
  cb.textContent = '×';
  cb.style.cssText = 'background:transparent;color:#fff;border:none;cursor:pointer;font-size:18px;padding:0 4px;line-height:1';
  cb.onclick = function(){
    bar.remove();
    document.querySelectorAll('[data-mh-label]').forEach(function(el){ el.remove(); });
    document.querySelectorAll('[data-mh-input]').forEach(function(el){
      el.style.outline = ''; el.style.outlineOffset = ''; el.removeAttribute('data-mh-input');
    });
    document.removeEventListener('keydown', kh, true);
  };
  bar.appendChild(cb);
  document.body.appendChild(bar);

  var kh = function(e){
    if(!document.getElementById('__mh_bar')) return;
    if(e.key === 'Enter' && !e.isComposing){
      e.preventDefault(); e.stopPropagation();
      go(e.shiftKey ? cur - 1 : cur + 1);
    } else if(e.key === 'Escape'){ cb.click(); }
  };
  document.addEventListener('keydown', kh, true);

  go(0);
  console.log('输入框命中：共 ' + allHits.length + ' 处，涉及 ' + document.querySelectorAll('[data-mh-input]').length + ' 个框');
})();
