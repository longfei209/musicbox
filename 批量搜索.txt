(function(){
  // 清理旧面板、旧标记、旧监听
  var old = document.getElementById('__mh_panel');
  if(old) old.remove();
  document.querySelectorAll('[data-mh-mark]').forEach(function(el){
    el.style.outline = ''; el.style.outlineOffset = ''; el.removeAttribute('data-mh-mark');
  });
  document.querySelectorAll('[data-mh-tag]').forEach(function(el){ el.remove(); });
  if(window.__mhUpdateTags){
    window.removeEventListener('scroll', window.__mhUpdateTags, true);
    window.removeEventListener('resize', window.__mhUpdateTags, true);
    window.__mhUpdateTags = null;
  }

  var panel = document.createElement('div');
  panel.id = '__mh_panel';
  panel.style.cssText = 'position:fixed;top:16px;right:16px;width:260px;background:#fff;border:1px solid #ddd;border-radius:10px;box-shadow:0 4px 20px rgba(0,0,0,.15);z-index:2147483647;font-family:system-ui;padding:12px;font-size:13px';
  panel.innerHTML =
    '<div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:8px">' +
      '<strong style="font-size:14px">输入框关键词标记</strong>' +
      '<button id="__mh_close" style="border:none;background:transparent;font-size:18px;cursor:pointer;line-height:1;padding:0 4px">×</button>' +
    '</div>' +
    '<textarea id="__mh_input" placeholder="每行一个词，或用空格/逗号分隔" style="width:100%;height:70px;box-sizing:border-box;padding:6px;border:1px solid #ccc;border-radius:6px;resize:vertical;font-size:13px;font-family:inherit"></textarea>' +
    '<div style="display:flex;gap:6px;margin-top:8px">' +
      '<button id="__mh_run" style="flex:1;padding:6px;border:none;border-radius:6px;background:#0067c0;color:#fff;cursor:pointer;font-size:13px">标记</button>' +
      '<button id="__mh_clear" style="flex:1;padding:6px;border:none;border-radius:6px;background:#666;color:#fff;cursor:pointer;font-size:13px">清除</button>' +
    '</div>' +
    '<div id="__mh_result" style="margin-top:8px;font-size:12px;color:#666"></div>';
  document.body.appendChild(panel);

  var input = panel.querySelector('#__mh_input');
  var runBtn = panel.querySelector('#__mh_run');
  var clearBtn = panel.querySelector('#__mh_clear');
  var closeBtn = panel.querySelector('#__mh_close');
  var result = panel.querySelector('#__mh_result');

  var tags = [];

  function clearMarks(){
    document.querySelectorAll('[data-mh-mark]').forEach(function(el){
      el.style.outline = ''; el.style.outlineOffset = ''; el.removeAttribute('data-mh-mark');
    });
    document.querySelectorAll('[data-mh-tag]').forEach(function(el){ el.remove(); });
    tags = [];
  }

  function updateTags(){
    tags.forEach(function(t){
      var r = t.el.getBoundingClientRect();
      if(r.bottom < 0 || r.top > window.innerHeight){
        t.tag.style.display = 'none';
        return;
      }
      t.tag.style.display = '';
      t.tag.style.left = Math.max(4, r.left) + 'px';
      t.tag.style.top = Math.max(4, r.top - 20) + 'px';
    });
  }

  function run(){
    clearMarks();
    var kws = [...new Set(
      input.value.split(/[\s,，、\n]+/).map(function(s){return s.trim()}).filter(Boolean)
    )];
    if(!kws.length){ result.textContent = '请输入关键词'; return; }

    var skip = ['hidden','password','file','checkbox','radio','submit','button','reset','image','range','color','date','time','month','week'];
    var boxCount = 0;

    document.querySelectorAll('input, textarea').forEach(function(el){
      if(el.closest('#__mh_panel')) return;
      var type = (el.type || 'text').toLowerCase();
      if(skip.indexOf(type) !== -1 || el.disabled) return;
      var v = el.value || '';
      if(!v) return;

      var lv = v.toLowerCase();
      var matched = kws.filter(function(kw){ return lv.indexOf(kw.toLowerCase()) !== -1; });
      if(!matched.length) return;

      el.setAttribute('data-mh-mark','1');
      el.style.outline = '2px solid #ff8a00';
      el.style.outlineOffset = '1px';

      var tag = document.createElement('div');
      tag.setAttribute('data-mh-tag','1');
      tag.textContent = matched.join(' · ');
      tag.style.cssText = 'position:fixed;background:#ff8a00;color:#fff;font-size:11px;line-height:16px;padding:0 6px;border-radius:8px;z-index:2147483646;pointer-events:none;font-family:system-ui;white-space:nowrap;max-width:220px;overflow:hidden;text-overflow:ellipsis';
      document.body.appendChild(tag);
      tags.push({ el: el, tag: tag });

      boxCount++;
    });

    updateTags();
    window.addEventListener('scroll', updateTags, true);
    window.addEventListener('resize', updateTags, true);
    window.__mhUpdateTags = updateTags;

    result.textContent = '共 ' + boxCount + ' 个框匹配';
  }

  runBtn.onclick = run;
  clearBtn.onclick = function(){ clearMarks(); result.textContent = ''; };
  closeBtn.onclick = function(){
    clearMarks();
    if(window.__mhUpdateTags){
      window.removeEventListener('scroll', window.__mhUpdateTags, true);
      window.removeEventListener('resize', window.__mhUpdateTags, true);
      window.__mhUpdateTags = null;
    }
    panel.remove();
  };

  input.focus();
})();
